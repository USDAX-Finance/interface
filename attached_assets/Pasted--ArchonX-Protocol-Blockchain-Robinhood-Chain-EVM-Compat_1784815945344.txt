## 📖 Overview

**ArchonX** is a dual-token DeFi protocol that combines an overcollateralized stablecoin with a governance token.

| Token | Ticker | Type | Target Price |
| :--- | :--- | :--- | :--- |
| **Stablecoin** | `USDAX` | USD Pegged (1:1) | $1.00 |
| **Governance Token** | `AKX` | ERC-20 Votes (Floating) | Market Driven (Low Cap) |

**Core Features**:
- Mint `USDAX` by depositing overcollateralized assets (WETH/WBTC).
- Earn variable APY by staking `AKX`.
- Liquidation mechanism with a 10% bonus for liquidators.
- 7-day cooldown for unstaking to ensure protocol stability.

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";

        _;

    function mint(address to, uint256 amount) external onlyMinter {
        _mint(to, amount);
    }

    function burn(address from, uint256 amount) external onlyMinter {
        _burn(from, amount);
    }

    function addMinter(address minter) external onlyOwner {
        minters[minter] = true;
    }

    function removeMinter(address minter) external onlyOwner {
        minters[minter] = false;
    }
}

    {
        super._afterTokenTransfer(from, to, amount);
    }

        require(totalSupply() + amount <= MAX_SUPPLY, "Exceeds max supply");
        super._mint(to, amount);
    }

3.3 contracts/OracleLib.sol (Chainlink Integration)
solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

import "@chainlink/contracts/src/v0.8/interfaces/AggregatorV3Interface.sol";

contract OracleLib {
    mapping(address => address) public priceFeeds;
    uint256 public constant TIMEOUT = 3 hours;

    function getPrice(address token) external view returns (uint256) {
        address feed = priceFeeds[token];
        require(feed != address(0), "Price feed not set");

        AggregatorV3Interface priceFeed = AggregatorV3Interface(feed);
        (, int256 price, , uint256 updatedAt, ) = priceFeed.latestRoundData();

        require(updatedAt >= block.timestamp - TIMEOUT, "Stale price!");
        require(price > 0, "Invalid price");

        // Chainlink 8 decimals -> 18 decimals
        return uint256(price) * 1e10;
    }

    function setPriceFeed(address token, address feed) external {
        priceFeeds[token] = feed;
    }
}
3.4 contracts/DSCEngine.sol (Core Logic)
solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/security/ReentrancyGuard.sol";
import "@openzeppelin/contracts/access/Ownable.sol";
import "./USDAX.sol";
import "./OracleLib.sol";

contract DSCEngine is ReentrancyGuard, Ownable {
    USDAX public immutable usdax;
    OracleLib public immutable oracle;

    uint256 public constant LIQUIDATION_THRESHOLD = 80;
    uint256 public constant LIQUIDATION_BONUS = 10;
    uint256 public constant MIN_HEALTH_FACTOR = 1e18;

    mapping(address => mapping(address => uint256)) public collateralDeposited;
    mapping(address => uint256) public usdaxMinted;
    mapping(address => uint256) public totalCollateralValue;

    address[] public collateralTokens;
    mapping(address => bool) public isCollateralToken;

    event CollateralDeposited(address indexed user, address token, uint256 amount);
    event USDAXMinted(address indexed user, uint256 amount);
    event CollateralRedeemed(address indexed user, address token, uint256 amount);
    event USDAXBurned(address indexed user, uint256 amount);
    event Liquidated(address indexed user, address liquidator, uint256 amount);

    constructor(address _usdax, address _oracle) {
        usdax = USDAX(_usdax);
        oracle = OracleLib(_oracle);
    }

    modifier onlyCollateralToken(address token) {
        require(isCollateralToken[token], "Not collateral token");
        _;
    }

    function depositCollateralAndMintUSDX(
        address tokenCollateralAddress,
        uint256 amountCollateral,
        uint256 amountUSDXToMint
    ) external nonReentrant onlyCollateralToken(tokenCollateralAddress) {
        IERC20(tokenCollateralAddress).transferFrom(
            msg.sender,
            address(this),
            amountCollateral
        );
        collateralDeposited[msg.sender][tokenCollateralAddress] += amountCollateral;
        _mintUSDX(msg.sender, amountUSDXToMint);
        _updateTotalCollateralValue(msg.sender);
        _revertIfHealthFactorIsBroken(msg.sender);
    }

    function redeemCollateralAndBurnUSDX(
        address tokenCollateralAddress,
        uint256 amountCollateral,
        uint256 amountUSDXToBurn
    ) external nonReentrant onlyCollateralToken(tokenCollateralAddress) {
        _burnUSDX(msg.sender, amountUSDXToBurn);
        collateralDeposited[msg.sender][tokenCollateralAddress] -= amountCollateral;
        IERC20(tokenCollateralAddress).transfer(msg.sender, amountCollateral);
        _updateTotalCollateralValue(msg.sender);
        _revertIfHealthFactorIsBroken(msg.sender);
    }

    function liquidate(
        address user,
        address tokenCollateralAddress,
        uint256 debtToCover
    ) external nonReentrant onlyCollateralToken(tokenCollateralAddress) {
        require(getHealthFactor(user) < MIN_HEALTH_FACTOR, "User is healthy");

        uint256 collateralValue = _getCollateralValue(
            tokenCollateralAddress,
            collateralDeposited[user][tokenCollateralAddress]
        );
        uint256 bonusCollateral = (debtToCover * LIQUIDATION_BONUS) / 100;
        uint256 totalCollateralToTake = debtToCover + bonusCollateral;

        collateralDeposited[user][tokenCollateralAddress] -= totalCollateralToTake;
        IERC20(tokenCollateralAddress).transfer(msg.sender, totalCollateralToTake);
        _burnUSDX(msg.sender, debtToCover);
        _updateTotalCollateralValue(user);

        emit Liquidated(user, msg.sender, debtToCover);
    }

    function _mintUSDX(address user, uint256 amount) internal {
        usdaxMinted[user] += amount;
        usdax.mint(user, amount);
        emit USDAXMinted(user, amount);
    }

    function _burnUSDX(address user, uint256 amount) internal {
        usdaxMinted[user] -= amount;
        usdax.burn(user, amount);
        emit USDAXBurned(user, amount);
    }

    function _getCollateralValue(address token, uint256 amount) internal view returns (uint256) {
        uint256 price = oracle.getPrice(token);
        return (price * amount) / 1e18;
    }

    function _updateTotalCollateralValue(address user) internal {
        uint256 total;
        for (uint256 i = 0; i < collateralTokens.length; i++) {
            address token = collateralTokens[i];
            uint256 amount = collateralDeposited[user][token];
            total += _getCollateralValue(token, amount);
        }
        totalCollateralValue[user] = total;
    }

    function _revertIfHealthFactorIsBroken(address user) internal view {
        require(getHealthFactor(user) >= MIN_HEALTH_FACTOR, "Health factor broken");
    }

    function getHealthFactor(address user) public view returns (uint256) {
        uint256 collateralValue = totalCollateralValue[user];
        uint256 debtValue = usdaxMinted[user]; // 1:1 with USD
        if (debtValue == 0) return type(uint256).max;
        return (collateralValue * 1e18) / debtValue;
    }

    function addCollateralToken(address token) external onlyOwner {
        require(!isCollateralToken[token], "Already added");
        isCollateralToken[token] = true;
        collateralTokens.push(token);
    }
}
3.5 contracts/Staking.sol (APY & Rewards)
solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/security/ReentrancyGuard.sol";

contract Staking is ReentrancyGuard {
    IERC20 public immutable stakingToken;
    IERC20 public immutable rewardToken;

    uint256 public constant COOLDOWN_PERIOD = 7 days;
    uint256 public baseAPY = 15;

    struct Stake {
        uint256 amount;
        uint256 timestamp;
        uint256 lastRewardTime;
        uint256 pendingRewards;
    }

    mapping(address => Stake) public stakes;
    uint256 public totalStaked;

    event Staked(address indexed user, uint256 amount);
    event Unstaked(address indexed user, uint256 amount);
    event RewardsClaimed(address indexed user, uint256 amount);

    constructor(address _stakingToken, address _rewardToken) {
        stakingToken = IERC20(_stakingToken);
        rewardToken = IERC20(_rewardToken);
    }

    function stake(uint256 amount) external nonReentrant {
        require(amount > 0, "Cannot stake 0");
        _claimRewards(msg.sender);
        stakingToken.transferFrom(msg.sender, address(this), amount);

        Stake storage userStake = stakes[msg.sender];
        userStake.amount += amount;
        userStake.timestamp = block.timestamp;
        userStake.lastRewardTime = block.timestamp;
        totalStaked += amount;
        emit Staked(msg.sender, amount);
    }

    function unstake(uint256 amount) external nonReentrant {
        Stake storage userStake = stakes[msg.sender];
        require(userStake.amount >= amount, "Insufficient stake");
        require(block.timestamp >= userStake.timestamp + COOLDOWN_PERIOD, "Cooldown not met");

        _claimRewards(msg.sender);
        userStake.amount -= amount;
        totalStaked -= amount;
        stakingToken.transfer(msg.sender, amount);
        emit Unstaked(msg.sender, amount);
    }

    function claimRewards() external nonReentrant {
        _claimRewards(msg.sender);
    }

    function _claimRewards(address user) internal {
        Stake storage userStake = stakes[user];
        uint256 rewards = _calculateRewards(user);
        if (rewards > 0) {
            userStake.pendingRewards = 0;
            userStake.lastRewardTime = block.timestamp;
            rewardToken.transfer(user, rewards);
            emit RewardsClaimed(user, rewards);
        }
    }

    function _calculateRewards(address user) internal view returns (uint256) {
        Stake storage userStake = stakes[user];
        if (userStake.amount == 0) return 0;

        uint256 timeElapsed = block.timestamp - userStake.lastRewardTime;
        uint256 rewardRate = (userStake.amount * baseAPY * 1e18) / (365 days * 100);
        return (rewardRate * timeElapsed) / 1e18;
    }

    function getPendingRewards(address user) external view returns (uint256) {
        return _calculateRewards(user) + stakes[user].pendingRewards;
    }
}

const hre = require("hardhat");

async function main() {
  console.log("Deploying ArchonX Protocol to Robinhood Testnet...");

  const usdax = await USDAX.deploy();

  console.log(`✅ USDAX deployed to: ${await usdax.getAddress()}`);

  // 2. Deploy AKX
  const AKX = await hre.ethers.getContractFactory("AKX");
  const akx = await AKX.deploy();
  await akx.waitForDeployment();
  console.log(`✅ AKX deployed to: ${await akx.getAddress()}`);

  // 3. Deploy OracleLib
  const OracleLib = await hre.ethers.getContractFactory("OracleLib");
  const oracle = await OracleLib.deploy();
  await oracle.waitForDeployment();
  console.log(`✅ OracleLib deployed to: ${await oracle.getAddress()}`);

  // 4. Set Price Feed (WETH/USD on Testnet)
  // Note: Using a well-known testnet feed. For Robinhood, we use the ETH/USD feed from Sepolia/Ethereum testnets.
  // Replace with actual feed address available on Robinhood testnet if different.
  const WETH_FEED = "0x5f4eC3Df9cbd43714FE2740f5E3616155c5b8419"; // Mainnet feed, usually works on EVM testnets
  const WETH_ADDRESS = "0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2";
  await oracle.setPriceFeed(WETH_ADDRESS, WETH_FEED);
  console.log(`✅ Oracle price feed set for WETH`);

  // 5. Deploy DSCEngine
  const DSCEngine = await hre.ethers.getContractFactory("DSCEngine");
  const engine = await DSCEngine.deploy(await usdax.getAddress(), await oracle.getAddress());
  await engine.waitForDeployment();
  console.log(`✅ DSCEngine deployed to: ${await engine.getAddress()}`);

  // 6. Grant Minter Role to Engine
  await usdax.addMinter(await engine.getAddress());
  console.log(`✅ Minter role granted to DSCEngine`);

  // 7. Add WETH as Collateral Token
  await engine.addCollateralToken(WETH_ADDRESS);
  console.log(`✅ WETH added as collateral token`);

  // 8. Deploy Staking
  const Staking = await hre.ethers.getContractFactory("Staking");
  const staking = await Staking.deploy(await akx.getAddress(), await akx.getAddress());
  await staking.waitForDeployment();
  console.log(`✅ Staking deployed to: ${await staking.getAddress()}`);

  console.log("\n🏛️ ArchonX Protocol Deployment Complete!");
  console.log(`📊 USDAX: ${await usdax.getAddress()}`);
  console.log(`📊 AKX: ${await akx.getAddress()}`);
  console.log(`📊 DSCEngine: ${await engine.getAddress()}`);
  console.log(`📊 Staking: ${await staking.getAddress()}`);
}

  process.exitCode = 1;
